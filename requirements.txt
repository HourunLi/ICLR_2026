# CLIR is implemented with SWIFT (https://github.com/aster2024/SWIFT) as an
# architectural reference, so the core numerical stack is pinned to match
# SWIFT's requirements.txt. This avoids subtle behavior/serialization drift
# between whatever future SWIFT-aligned hidden-state extraction pipeline
# produces and the model code in this repo.
#
# Tier 1 is everything the current code in this repo actually imports
# (src/, train_clir.py, score_clir.py, tests/). Install this to run the
# smoke tests and the toy end-to-end pipeline.
torch==2.3.1
numpy==2.2.6
pytest==8.3.3

# Tier 2 is not imported by anything in this repo yet. It matches what
# docs/handoff.md lists as the P0 task: a script that loads a real LLM and
# extracts per-token hidden states the way SWIFT's train/extract_train.py
# and eval/get_rewards.py do (both import AutoTokenizer/AutoModelForCausalLM
# from transformers). Uncomment when that script exists.
#
# NOTE: SWIFT's own requirements.txt pins accelerate==0.32.1 together with
# numpy==2.2.6, but accelerate==0.32.1 requires numpy<2.0.0 -- those two
# pins conflict and `pip install -r requirements.txt` on SWIFT's own file
# fails dependency resolution as written. accelerate started supporting
# numpy>=2.0 well before 1.x, so this file pins a newer accelerate instead
# of copying SWIFT's exact 0.32.1.
# transformers==4.51.2
# accelerate>=1.0.0
